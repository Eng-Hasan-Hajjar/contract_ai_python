    def _warn_mismatch(self, detect):
        """يُظهر تحذير عدم التوافق ويعطي المستخدم خيار التغيير."""
        from tkinter import messagebox as mb
        answer = mb.askyesno(
            "⚠  تحذير: نوع العقد قد لا يكون صحيحاً",
            f"{detect.warning_msg}\n\n{detect.suggestion}",
            icon="warning"
        )
        if answer:
            # المستخدم يريد تغيير النوع → ارجع للرئيسية
            self.nav_cb("home")
